"use client";

import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
import { Clock, Play, Zap } from "lucide-react";
import { Switch } from "@/components/approvals/switch";
import { runEscalationAction, setEscalationAction } from "../extra-actions";

type Rule = { id: string; code: string; name: string; days: number | null };

export function EscalationSettings({ rules }: { rules: Rule[] }) {
  const router = useRouter();
  const [defaultDays, setDefaultDays] = useState("3");
  const [edits, setEdits] = useState<Record<string, string>>({});
  const [msg, setMsg] = useState<{ ok: boolean; text: string } | null>(null);
  const [pending, start] = useTransition();
  const anyOn = rules.some((r) => r.days !== null);

  async function apply(list: { id: string; days: string }[]) {
    for (const x of list) {
      const res = await setEscalationAction(x.id, x.days);
      if (!res.ok) return res.error;
    }
    return null;
  }

  function validDefault() {
    const n = Number(defaultDays);
    if (!Number.isInteger(n) || n < 1 || n > 60) {
      setMsg({ ok: false, text: "Days must be a whole number between 1 and 60" });
      return false;
    }
    return true;
  }

  function master(on: boolean) {
    setMsg(null);
    if (on && !validDefault()) return;
    start(async () => {
      const targets = on
        ? rules.filter((r) => r.days === null).map((r) => ({ id: r.id, days: defaultDays }))
        : rules.filter((r) => r.days !== null).map((r) => ({ id: r.id, days: "" }));
      const err = await apply(targets);
      if (err) setMsg({ ok: false, text: err });
      else {
        setEdits({});
        router.refresh();
      }
    });
  }

  function one(r: Rule, days: string) {
    setMsg(null);
    if (days !== "") {
      const n = Number(days);
      if (!Number.isInteger(n) || n < 1 || n > 60) {
        setMsg({ ok: false, text: "Days must be a whole number between 1 and 60" });
        return;
      }
    }
    start(async () => {
      const err = await apply([{ id: r.id, days }]);
      if (err) setMsg({ ok: false, text: err });
      else {
        setEdits((p) => {
          const n = { ...p };
          delete n[r.id];
          return n;
        });
        router.refresh();
      }
    });
  }

  function runNow() {
    setMsg(null);
    start(async () => {
      const res = await runEscalationAction();
      if (!res.ok) setMsg({ ok: false, text: res.error });
      else setMsg({ ok: true, text: `Escalation check done: ${res.data.escalated} request(s) escalated` });
      if (res.ok) router.refresh();
    });
  }

  return (
    <section id="escalation" className="rounded-card border bg-card p-6 shadow-card">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex items-center gap-3">
          <Zap className="h-8 w-8 text-blue-600" fill="currentColor" fillOpacity={0.2} />
          <div>
            <h2 className="font-heading text-xl font-semibold text-navy-800">Escalation</h2>
            <p className="text-sm text-slate-500">Set up escalation rules for pending approvals</p>
          </div>
        </div>
        <label className="flex items-center gap-3 text-sm font-semibold text-navy-800">
          Enable Escalation
          <Switch checked={anyOn} onChange={master} disabled={pending || rules.length === 0} label="Enable escalation for all rules" />
        </label>
      </div>

      <div className="mt-4 flex flex-wrap items-center gap-2 rounded-xl bg-blue-50/70 px-4 py-3 text-sm text-navy-800">
        <Clock className="h-5 w-5 text-blue-600" />
        If a request waits more than
        <input
          value={defaultDays}
          onChange={(e) => setDefaultDays(e.target.value)}
          inputMode="numeric"
          aria-label="Default days"
          className="h-8 w-14 rounded-md border bg-white px-2 text-center text-sm outline-none focus:ring-2 focus:ring-blue-200"
        />
        days at a step, the next level is notified. Turn the switch off = off.
      </div>

      {rules.length === 0 ? (
        <p className="mt-4 text-sm text-muted-foreground">No rules yet. Save a rule above first.</p>
      ) : (
        <ul className="mt-4 divide-y">
          {rules.map((r) => {
            const shown = edits[r.id] ?? String(r.days ?? "");
            const dirty = edits[r.id] !== undefined && edits[r.id] !== String(r.days ?? "");
            return (
              <li key={r.id} className="flex flex-wrap items-center gap-3 py-3 text-sm">
                <span className="min-w-48 flex-1">
                  <span className="font-semibold text-navy-800">{r.name}</span>
                  <span className="ml-2 text-xs text-slate-500">{r.code}</span>
                </span>
                <Switch
                  checked={r.days !== null}
                  disabled={pending}
                  label={`Escalation for ${r.name}`}
                  onChange={(on) => (on ? (validDefault() ? one(r, defaultDays) : undefined) : one(r, ""))}
                />
                <input
                  value={shown}
                  disabled={r.days === null}
                  onChange={(e) => setEdits((p) => ({ ...p, [r.id]: e.target.value }))}
                  inputMode="numeric"
                  aria-label={`Escalation days for ${r.name}`}
                  className="h-9 w-16 rounded-md border bg-white px-2 text-center outline-none focus:ring-2 focus:ring-blue-200 disabled:bg-slate-100"
                />
                <span className="text-slate-500">days</span>
                {dirty && (
                  <button
                    type="button"
                    disabled={pending}
                    onClick={() => one(r, shown)}
                    className="rounded-md bg-primary px-3 py-1.5 text-xs font-semibold text-primary-foreground disabled:opacity-60"
                  >
                    Save
                  </button>
                )}
              </li>
            );
          })}
        </ul>
      )}

      <div className="mt-4 flex flex-wrap items-center gap-3">
        <button
          type="button"
          disabled={pending}
          onClick={runNow}
          className="inline-flex h-10 items-center gap-2 rounded-full border bg-card px-5 text-sm font-semibold text-navy-800 hover:bg-slate-50 disabled:opacity-60"
        >
          <Play className="h-4 w-4" fill="currentColor" /> Run escalation check now
        </button>
        {msg && <p className={`text-sm ${msg.ok ? "text-emerald-700" : "text-destructive"}`}>{msg.text}</p>}
      </div>
    </section>
  );
}

export function RuleForm({ roles }: { roles: { id: string; name: string }[] }) {

        maxAmount: get("maxAmount"),

"use client";

import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
import { Button } from "@/components/ui/button";
import { cancelApprovalAction, decideApprovalAction } from "./actions";

export function ApprovalButtons({ id, mode }: { id: string; mode: "decide" | "cancel" }) {
  const router = useRouter();
  const [error, setError] = useState("");
  const [pending, start] = useTransition();

  function run(fn: () => Promise<{ ok: boolean; error?: string }>) {
    setError("");
    start(async () => {
      const res = await fn();
      if (!res.ok) setError(res.error ?? "Failed");
      else router.refresh();
    });
  }

  function approve() {
    run(() => decideApprovalAction(id, "APPROVED", ""));
  }

  function reject() {
    const reason = window.prompt("Why is this being rejected?");
    if (reason === null) return;
    if (reason.trim().length < 3) {
      setError("A reason is required (at least 3 characters)");
      return;
    }
    run(() => decideApprovalAction(id, "REJECTED", reason));
  }

  function cancel() {
    if (!window.confirm("Cancel this request?")) return;
    run(() => cancelApprovalAction(id));
  }

  return (
    <div className="space-y-1">
      {mode === "decide" ? (
        <div className="flex gap-1">
          <Button size="sm" disabled={pending} onClick={approve}>
            Approve
          </Button>
          <Button size="sm" variant="destructive" disabled={pending} onClick={reject}>
            Reject
          </Button>
        </div>
      ) : (
        <Button size="sm" variant="outline" disabled={pending} onClick={cancel}>
          Cancel
        </Button>
      )}
      {error && <p className="text-xs text-destructive">{error}</p>}
    </div>
  );
}

"use client";

import { useState, useTransition } from "react";
import { useRouter } from "next/navigation";
import { Switch } from "@/components/approvals/switch";
import { setWorkflowActiveAction } from "./extra-actions";

export function RuleToggle({ id, active, name }: { id: string; active: boolean; name: string }) {
  const router = useRouter();
  const [on, setOn] = useState(active);
  const [err, setErr] = useState("");
  const [pending, start] = useTransition();

  function change(next: boolean) {
    setOn(next);
    setErr("");
    start(async () => {
      const res = await setWorkflowActiveAction(id, next);
      if (!res.ok) {
        setOn(!next);
        setErr(res.error);
      } else {
        router.refresh();
      }
    });
  }

  return (
    <div className="flex items-center gap-2">
      <Switch checked={on} onChange={change} disabled={pending} label={`${name} active`} />
      {err && <span className="text-xs text-destructive">{err}</span>}
    </div>
  );
}

import { useState, useTransition } from "react";
import { markAllNotificationsReadAction, markNotificationReadAction } from "@/app/(app)/actions";
import { cn } from "@/lib/utils";

type Item = {
  id: string;
  title: string;
  message: string;
  href: string | null;
  read: boolean;
  createdAt: string;
};

export function NotificationList({ items }: { items: Item[] }) {
  const router = useRouter();
  const [list, setList] = useState(items);
  const [, start] = useTransition();
  const unread = list.filter((n) => !n.read).length;

  function open(n: Item) {
    if (!n.read) {
      setList((l) => l.map((x) => (x.id === n.id ? { ...x, read: true } : x)));
      start(async () => {
        await markNotificationReadAction(n.id);
      });
    }
    if (n.href) router.push(n.href);
  }

  function readAll() {
    setList((l) => l.map((x) => ({ ...x, read: true })));
    start(async () => {
      await markAllNotificationsReadAction();
    });
  }

  return (
    <div>
      {unread > 0 && (
        <div className="flex items-center justify-between border-b px-4 py-2 text-sm">
          <span className="text-muted-foreground">{unread} unread</span>
          <button onClick={readAll} className="text-denim-700 hover:underline">
            Mark all read
          </button>
        </div>
      )}
      {list.length === 0 ? (
        <p className="p-6 text-center text-sm text-muted-foreground">No notifications yet</p>
      ) : (
        <ul>
          {list.map((n) => (
            <li key={n.id}>
              <button
                onClick={() => open(n)}
                className={cn("block w-full border-b px-4 py-2.5 text-left hover:bg-muted", !n.read && "bg-denim-100/40")}

"use client";

import { useState, useTransition } from "react";
import { setNotificationPreferenceAction } from "@/app/(app)/actions";

type Row = { type: string; inApp: boolean; email: boolean };

const label = (t: string) => t.replace(/_/g, " ").toLowerCase().replace(/^\w/, (c) => c.toUpperCase());

// à¦ªà§à¦°à¦¤à¦¿à¦Ÿà¦¿ à¦Ÿà¦¿à¦• à¦¦à¦¿à¦²à§‡à¦‡ à¦¸à¦¾à¦¥à§‡ à¦¸à¦¾à¦¥à§‡ à¦¸à§‡à¦­ à¦¹à¦¯à¦¼; à¦¬à§à¦¯à¦°à§à¦¥ à¦¹à¦²à§‡ à¦†à¦—à§‡à¦° à¦…à¦¬à¦¸à§à¦¥à¦¾à¦¯à¦¼ à¦«à§‡à¦°à§‡
export function PreferencesForm({ rows }: { rows: Row[] }) {
  const [state, setState] = useState(rows);
  const [error, setError] = useState("");
  const [pending, start] = useTransition();

  function toggle(type: string, field: "inApp" | "email") {
    const prev = state;
    const cur = state.find((r) => r.type === type)!;
    const next = { ...cur, [field]: !cur[field] };
    setState(state.map((r) => (r.type === type ? next : r)));
    setError("");
    start(async () => {
      const res = await setNotificationPreferenceAction(type, next.inApp, next.email);
      if (!res.ok) {
        setState(prev);
        setError(res.error);
      }
    });
  }

  return (
    <div className="space-y-2">
      <div className="grid grid-cols-[1fr_auto_auto] items-center gap-x-4 gap-y-2 text-sm">
        <span />
        <span className="text-xs font-medium uppercase text-muted-foreground">In-app</span>
        <span className="text-xs font-medium uppercase text-muted-foreground">Email</span>
        {state.map((r) => (
          <div key={r.type} className="contents">
            <span>{label(r.type)}</span>
            <input
              type="checkbox"
              checked={r.inApp}
              disabled={pending}
              onChange={() => toggle(r.type, "inApp")}
              aria-label={`${label(r.type)} in-app`}
            />
            <input
              type="checkbox"
              checked={r.email}
              disabled={pending}
              onChange={() => toggle(r.type, "email")}
              aria-label={`${label(r.type)} email`}
            />
          </div>
        ))}
      </div>
      {error && <p className="text-sm text-destructive">{error}</p>}
      <p className="text-xs text-muted-foreground">Email messages are queued now and will be sent once email delivery is connected.</p>
    </div>
  );
}

export default async function AppLayout({ children }: { children: React.ReactNode }) {
  // Only "not logged in" (AuthError) goes to /login. Any other error is shown, not hidden.
  const user = await requireUser().catch((e) => {
    if (e instanceof AuthError) return null;
    throw e;
  });
  if (!user) redirect("/login");

  const [company, companies, unreadCount, recent, me, role, avatar] = await Promise.all([
    prisma.company.findUnique({
      where: { id: user.companyId },
      select: { tradeName: true, legalName: true },
    }),
    listSwitchOptions(user.id),
    notificationService.unreadCount(),
    notificationService.list(false, 8),
    prisma.user.findUnique({ where: { id: user.id }, select: { name: true } }),
    prisma.role.findUnique({ where: { id: user.roleId }, select: { name: true } }),
    getAvatarUrl(user.id),
  ]);
  const textileEnabled = await isTextileEnabled(user.id);
  // à¦¯à§‡ permission à¦¨à§‡à¦‡, à¦¸à§‡à¦‡ à¦®à§‡à¦¨à§ à¦¦à§‡à¦–à¦¾à¦¨à§‹ à¦¹à¦¯à¦¼ à¦¨à¦¾
    const groups: NavGroupData[] = NAV.filter((g) => g.feature !== "textile" || textileEnabled).map((g) => ({
    label: g.label,
    items: g.items.filter((i) => !i.permission || can(user, i.permission)),
  })).filter((g) => g.items.length > 0);

  return (
    <AppShell
      groups={groups}
      companyName={company?.tradeName ?? company?.legalName ?? "-"}
      companies={companies.map((c) => ({ id: c.id, name: c.name }))}
      activeCompanyId={user.companyId}
      userEmail={user.email}
      userName={me?.name ?? user.email}
      userRole={role?.name ?? ""}
      avatarUrl={avatar}
      unreadCount={unreadCount}
      notifications={recent.map((n) => ({
        id: n.id,
        title: n.title,
        message: n.message,
        href: n.href ?? null,
        read: n.readAt !== null,
        createdAt: n.createdAt.toISOString(),
      }))}
    >
      {children}
    </AppShell>
  );
}

export function Sidebar({
  groups,
  open,
  onClose,
}: {
  groups: NavGroupData[];
  open: boolean;
  onClose: () => void;
}) {
  const pathname = usePathname();

  const readyItems = groups.flatMap((g) => g.items.filter((i) => i.ready));
  const top = readyItems.find((i) => i.href === PIN_TOP);
  const bottom = readyItems.find((i) => i.href === PIN_BOTTOM);

  // Dashboard ar Analytics group theke bad (alada pin kora), "soon" item-o bad
  const visible = groups
    .map((g) => ({
      ...g,
      items: g.items.filter((i) => i.ready && i.href !== PIN_TOP && i.href !== PIN_BOTTOM),
    }))
    .filter((g) => g.items.length > 0);

  // shobcheye lomba match-kora href-i active (orders vs orders/amendments)
  const activeHref = readyItems
    .map((i) => i.href)
    .filter((h) => pathname === h || pathname.startsWith(h + "/"))
    .sort((a, b) => b.length - a.length)[0];

  const activeGroup = visible.find((g) => g.items.some((i) => i.href === activeHref))?.label;

  const [openGroups, setOpenGroups] = useState<Set<string>>(new Set(visible.map((g) => g.label)));

  // page change hole oi page-er group auto-open
  const [prevActive, setPrevActive] = useState(activeGroup);
  if (activeGroup !== prevActive) {
    setPrevActive(activeGroup);
    if (activeGroup) setOpenGroups((p) => new Set(p).add(activeGroup));
  }

  function toggle(label: string) {
    setOpenGroups((prev) => {
      const next = new Set(prev);
      if (next.has(label)) next.delete(label);
      else next.add(label);
      return next;
    });
  }

  return (
    <>
      {/* à¦®à§‹à¦¬à¦¾à¦‡à¦²à§‡ à¦®à§‡à¦¨à§ à¦–à§à¦²à¦²à§‡ à¦ªà§‡à¦›à¦¨à§‡à¦° à¦à¦¾à¦ªà¦¸à¦¾ à¦…à¦‚à¦¶ */}
      {open && <div className="fixed inset-0 z-30 bg-black/40 md:hidden" onClick={onClose} />}

      <aside
        className={cn(
          "fixed inset-y-0 left-0 z-40 flex w-64 shrink-0 flex-col bg-linear-to-b from-navy-800 to-navy-950 text-slate-200 transition-transform md:sticky md:top-0 md:h-screen md:translate-x-0",
          open ? "translate-x-0" : "-translate-x-full"
        )}
      >
        {/* Logo */}
        <div className="flex h-16 shrink-0 items-center justify-between px-4">
          <div className="flex items-center gap-2.5">
            <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-linear-to-br from-sky-400 to-blue-600 font-heading text-lg font-bold text-white shadow-lg shadow-blue-950/50">
              M
            </span>
            <div className="leading-tight">
              <p className="font-heading text-base font-bold tracking-wide text-white">
                MERCH<span className="text-sky-300">FLOW</span>
              </p>
              <p className="text-[10px] text-slate-300/80">Garments &amp; Textile ERP</p>
            </div>
          </div>
          <button className="md:hidden" onClick={onClose} aria-label="Close menu">
            <X className="h-5 w-5" />
          </button>
        </div>

        {/* Dashboard: alada pin */}
        {top && (
          <div className="shrink-0 px-3 pb-2 pt-1">
            <NavLink item={top} active={top.href === activeHref} onClose={onClose} />
          </div>
        )}

        {/* Groups */}
        <nav className="flex-1 space-y-1 overflow-y-auto px-3 pb-4 pt-1">
          {visible.map((g) => {
            const isOpen = openGroups.has(g.label);
            return (
              <div key={g.label}>
                <button
                  type="button"
                  onClick={() => toggle(g.label)}
                  aria-expanded={isOpen}
                  className="flex w-full items-center justify-between rounded-md px-3 py-2 text-[11px] font-semibold uppercase tracking-wider text-sky-300/80 transition-colors hover:text-white"
                >
                  {g.label}
                  <ChevronDown className={cn("h-3.5 w-3.5 transition-transform", isOpen && "rotate-180")} />
                </button>
                
                <div
                  className={cn(
                    "grid transition-[grid-template-rows] duration-300 ease-out",
                    isOpen ? "grid-rows-[1fr]" : "grid-rows-[0fr]"
                  )}
                >
                  <ul className="min-h-0 space-y-0.5 overflow-hidden" inert={!isOpen}>
                    {g.items.map((item) => (
                      <li key={item.href}>
                        <NavLink item={item} active={item.href === activeHref} onClose={onClose} />
                      </li>
                    ))}
                    <li aria-hidden="true" className="h-2" />
                  </ul>
                </div>
              </div>
            );
          })}
        </nav>

        {/* Reports & Analytics: niche pin */}
